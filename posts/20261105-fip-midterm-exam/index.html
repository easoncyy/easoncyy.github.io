26.

def weigh(a: list, lo: int, hi: int) -> int:
    if lo == hi:
        return a[lo]
    mid = (lo + hi) // 2
    left = weigh(a, lo, mid)
    right = weigh(a, mid + 1, hi)
    if left >= right:
        chosen = left
    else:
        chosen = right
    print(chosen)
    return chosen

nums = [3, 8, 2, 6]
print(weigh(nums, 0, 3))
